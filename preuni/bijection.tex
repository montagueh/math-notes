\section{Bijection}
\subsection{Bijection Problem}
Consider $\ln : (0,\infty) \to \mathbb{R}$. Is it injective? Is it surjective? Is it bijective?
\begin{enumerate}
  \item Let $f(x) = e^{x}$ and $g(x) = \ln(x)$
    \begin{align*}
      (f \circ g) &= z \\
      z       &= e^{\ln(x)} \\
      \ln(z)  &= \ln(x) = t && \text{(adding var t)} \\
      e^{t}      &= z = x \\
      z       &= x \\
      (g \circ f) &= y \\
      y       &= \ln(e^{x}) \\
      e^{y}      &= e^{x} \\
      y       &= x
    \end{align*}
\end{enumerate}

    \begin{equation}
