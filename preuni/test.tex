  \section{B - proofs}
  \item \textbf{Proove that $2^{x}$ is injective. Name the technique that you are using:} 

    Proof (direct):
    Suppose,
    \begin{gather*}
      2^{a} = 2^{b} \\
      \ln(2^{a}) = \ln(2^{b}) \\
      a \ln(2) = b \ln(2) \\
      a = b
    \end{gather*}
    Because $2^{a} = 2^{b}$ leads to $a = b$ the function $2^{x}$ is injective.

  \section{C - transformations}
  \item \textbf{Estimate $\sin(0.03)$ without a calculator, and explain why in one sentence why sin $\theta < \theta$}
    At low values of sine, most of the change in arc length of the circle is in height, so $\sin(0.03) \approx 0.03$.

  \item \textbf{A wheel has centre height 5, radius 2, one full turn every 10 seconds, and the rider starts at
    the midline moving upward. Write $h(t)$, then find the height at $t = 2.5$s.}

    \[
      h = 6 \quad r = 2 \quad P = 10
    \]
    At $t = 2.5$
    \begin{align*}
      h(t) &= 2\sin \left(\frac{2\pi t}{10} \right) + 6 \\
           &= 2\sin \left(\frac{\pi}{2} \right) + 6 \\
           &= 2 + 6 \\
           &= 8
    \end{align*}
    

\end{enumerate}
